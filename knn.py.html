1 #!/usr/bin/env python3   2 """   3 knn: guess a row's disty from its k nearest rows.   4 Table 1: kernels flat (all neighbours count the same) and tri   5 (nearer counts more: weight 1 - d/dmax) at k=1..5, plus mu   6 (always guess the train mean). Table 2: flat kernel at k=4,   7 trained on 32..512 rows, or all, on BIGS --dir csvs with ROWS   8 rows. Both: 5 repeats of a 2-way cross-val, scored by RAE, on at   9 most --test test rows and --pool train rows.  10 (c) 2026, Tim Menzies <timm@ieee.org>, MIT license  11 """  12 import os, sys  # make hair importable from source or installed  13 sys.path.insert(0, os.path.dirname(os.path.dirname(  14                    os.path.abspath(__file__))))  15 from functools import partial  16 from hair import *  17   18 BEST_K = 4                        # table 1's pick, used by table 2  19 SIZES  = [32, 64, 128, 256, 512]  # train sizes for table 2  20 ROWS   = range(1000, 10000)       # table 2: sets with these #rows  21 BIGS   = 20                       # table 2: how many such sets  22   23 #-- kernels: guess row's disty from rows (nearest first) ------------  24 def flat(tbl, rows: list, row=None) -> float:  25   return mid(adds(disty(tbl, r) for r in rows))  26   27 def tri(tbl, rows: list, row: list) -> float:  28   ds = [distx(tbl, r, row) for r in rows]  29   ws = [1 - d/(max(ds) + TINY) for d in ds]  30   return (sum(w*disty(tbl, r) for w, r in zip(ws, rows))  31           / (sum(ws) + TINY))  32   33 #-- learner: treatment t = o(name, f, k, n); k,n None = all ------  34 def knn(tbl, train, test, ts: list):  35   "Learner: sort train once per test row; every t slices that sort."  36   for row in test:                     # xval shuffled train, so  37     want = disty(tbl, row)             # i < n = a random n rows  38     ns = sorted(range(len(train)),   39                 key= lambda i: distx(tbl, train[i], row))  40     for t in ts:  41       rows = [train[i] for i in ns if t.n is None or i < t.n]  42       yield t.name, want, t.f(tbl, rows[:t.k], row)  43   44 def main() -> None:  45   cli(the)                           46   print(conf(the, "knn.py", eval="rae", xval="5x2"))  47   print("# rae: 100 = no better than guessing the mean; lower=better\n"  48         "# ! marks a rank-1 (best) cell; last col = data set")  49   for x in [o(title = "kernels x k",   50               files = files(),  51               ts    = [o(name=f"{f.__name__}:k={k}", f=f, k=k, n=None)  52                        for f in (flat, tri) for k in range(1, the.k+1)]  53                       + [o(name="mu", f=flat, k=None, n=None)]  54               ),             55             o(title = f"flat:k={BEST_K} x train size (all = pool)",  56               files = csvs(the.dir, ROWS, BIGS, the.seed),  57               ts    = [o(name=f"n={n or 'all'}", f=flat, k=BEST_K, n=n)  58                        for n in SIZES + [None]]  59               )]:  60     print(f"\n# {x.title}\n")  61     report(learn(x.files, partial(knn, ts=x.ts), rae, n=2))  62   63 if __name__ == "__main__": main()  64
